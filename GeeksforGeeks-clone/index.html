busca_binaria.js
function buscaBinaria(lista, alvo) {
  let ini = 0, fim = lista.length - 1;
  while (ini <= fim) {
    const meio = (ini + fim) >> 1;
    if (lista[meio] === alvo) return meio;
    lista[meio] < alvo ? ini = meio + 1
                       : fim = meio - 1;
  }
  return -1; // O(log n)
}
